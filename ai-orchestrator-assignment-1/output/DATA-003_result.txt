import os
import tempfile
import csv

def count_csv_rows(filepath: str, skip_header: bool = True) -> int:
    if not os.path.exists(filepath):
        raise FileNotFoundError(f"The file {filepath} does not exist.")
    
    with open(filepath, 'r', newline='') as csvfile:
        reader = csv.reader(csvfile)
        rows = list(reader)
        
        if not rows:
            raise ValueError("The file is empty.")
        
        return len(rows) - (1 if skip_header else 0)

if __name__ == "__main__":
    csv_data = "header1,header2\nvalue1,value2\nvalue3,value4"
    with tempfile.NamedTemporaryFile(delete=False, mode='w', newline='') as temp_file:
        temp_file.write(csv_data)
        temp_file_path = temp_file.name
    
    try:
        row_count = count_csv_rows(temp_file_path)
        print(f"Number of data rows: {row_count}")
    finally:
        os.remove(temp_file_path)
